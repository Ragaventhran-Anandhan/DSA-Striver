public static String infixToPostfix(String s) {
    StringBuilder out = new StringBuilder();
    Deque<Character> st = new ArrayDeque<>();
    for (char c : s.toCharArray()) {
        if (Character.isLetterOrDigit(c)) out.append(c);
        else if (c == '(') st.push(c);
        else if (c == ')') {
            while (!st.isEmpty() && st.peek() != '(') out.append(st.pop());

            st.push(c);

    return out.toString();
}

public static int sumSubarrayMins(int[] arr) {
    int n = arr.length, MOD = 1_000_000_007;
    int[] left = new int[n], right = new int[n];
    Deque<Integer> st = new ArrayDeque<>();
    for (int i = 0; i < n; i++) {
        while (!st.isEmpty() && arr[st.peek()] > arr[i]) st.pop();
        left[i] = st.isEmpty() ? i + 1 : i - st.peek();
        st.push(i);
    }
    st.clear();
    for (int i = n - 1; i >= 0; i--) {
        while (!st.isEmpty() && arr[st.peek()] >= arr[i]) st.pop();
        right[i] = st.isEmpty() ? n - i : st.peek() - i;
        st.push(i);
    }
    long sum = 0;
    for (int i = 0; i < n; i++) sum = (sum + (long) arr[i] * left[i] % MOD * right[i]) % MOD;
    return (int) sum;
}

public static void lfuCache() {
    class LFU {
        int cap, min = 0;
        Map<Integer, Integer> val = new HashMap<>(), freq = new HashMap<>();
        Map<Integer, LinkedHashSet<Integer>> buckets = new HashMap<>();
        LFU(int cap) { this.cap = cap; }
        void touch(int k) {
            int f = freq.get(k);
            buckets.get(f).remove(k);
            if (f == min && buckets.get(f).isEmpty()) min++;
            freq.put(k, f + 1);
            buckets.computeIfAbsent(f + 1, x -> new LinkedHashSet<>()).add(k);
        }
        int get(int k) {
            if (!val.containsKey(k)) return -1;
            touch(k);
            return val.get(k);
        }
        void put(int k, int v) {
            if (cap == 0) return;
            if (val.containsKey(k)) { val.put(k, v); touch(k); return; }
            if (val.size() == cap) {
                int old = buckets.get(min).iterator().next();
                buckets.get(min).remove(old);
                val.remove(old); freq.remove(old);
            }
            val.put(k, v); freq.put(k, 1);
            buckets.computeIfAbsent(1, x -> new LinkedHashSet<>()).add(k);
            min = 1;
        }
    }
    LFU c = new LFU(2);
    c.put(1, 1); c.put(2, 2);
    System.out.println(c.get(1));
    c.put(3, 3);
    System.out.println(c.get(2));
}
