private static boolean dfs(char[][] b, String w, int i, int j, int k) {
    if (k == w.length()) return true;
    if (i < 0 || j < 0 || i >= b.length || j >= b[0].length || b[i][j] != w.charAt(k)) return false;

    b[i][j] = t;
    return ok;

public static String getPermutation(int n, int k) {
    List<Integer> nums = new ArrayList<>();
    int fact = 1;
    for (int i = 1; i < n; i++) { fact *= i; nums.add(i); }
    nums.add(n);
    k--;
    StringBuilder sb = new StringBuilder();
    while (true) {
        sb.append(nums.remove(k / fact));
        if (nums.isEmpty()) break;
        k %= fact;
        fact /= nums.size();
    }
    return sb.toString();
}

private static void dfs(int[][] m, int i, int j, int n, boolean[][] vis, StringBuilder p, List<String> res) {
    if (i == n - 1 && j == n - 1) { res.add(p.toString()); return; }
    vis[i][j] = true;
    int[] di = {1, 0, 0, -1}, dj = {0, -1, 1, 0};
    char[] c = {'D', 'L', 'R', 'U'};
    for (int d = 0; d < 4; d++) {
        int x = i + di[d], y = j + dj[d];
        if (x >= 0 && y >= 0 && x < n && y < n && m[x][y] == 1 && !vis[x][y]) {
            p.append(c[d]);
            dfs(m, x, y, n, vis, p, res);
            p.deleteCharAt(p.length() - 1);
        }
    }
    vis[i][j] = false;
}

private static void dfs(String s, int t, int i, String expr, long val, long last, List<String> res) {
    if (i == s.length()) { if (val == t) res.add(expr); return; }
    for (int j = i; j < s.length(); j++) {
        if (j > i && s.charAt(i) == '0') break;
        long cur = Long.parseLong(s.substring(i, j + 1));
        String part = s.substring(i, j + 1);
        if (i == 0) dfs(s, t, j + 1, part, cur, cur, res);
        else {
            dfs(s, t, j + 1, expr + "+" + part, val + cur, cur, res);
            dfs(s, t, j + 1, expr + "-" + part, val - cur, -cur, res);
            dfs(s, t, j + 1, expr + "*" + part, val - last + last * cur, last * cur, res);
        }
    }

public static List<Integer> kthPermutation(int[] arr, int k) {
    List<Integer> nums = new ArrayList<>();
    for (int x : arr) nums.add(x);
    Collections.sort(nums);
    int n = nums.size();
    int[] fact = new int[n + 1];
    fact[0] = 1;
    for (int i = 1; i <= n; i++) fact[i] = fact[i - 1] * i;
    k--;
    List<Integer> res = new ArrayList<>();
    for (int i = n; i >= 1; i--) {
        int idx = k / fact[i - 1];
        res.add(nums.remove(idx));
        k %= fact[i - 1];
    }
    return res;
}

private static void place(int r, int n, int[] pos, boolean[] col, boolean[] d1, boolean[] d2, List<List<String>> res) {
    if (r == n) {
        List<String> b = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            char[] row = new char[n];
            Arrays.fill(row, '.');
            row[pos[i]] = 'Q';
            b.add(new String(row));
        }
        res.add(b);
        return;
    }
    for (int c = 0; c < n; c++) {
        if (col[c] || d1[r - c + n] || d2[r + c]) continue;
        pos[r] = c; col[c] = d1[r - c + n] = d2[r + c] = true;
        place(r + 1, n, pos, col, d1, d2, res);
        col[c] = d1[r - c + n] = d2[r + c] = false;
    }
}

private static boolean solve(char[][] b) {
    for (int i = 0; i < 9; i++)
        for (int j = 0; j < 9; j++) {
            if (b[i][j] != '.') continue;
            for (char c = '1'; c <= '9'; c++) {
                if (ok(b, i, j, c)) {
                    b[i][j] = c;
                    if (solve(b)) return true;
                    b[i][j] = '.';
                }
            }
            return false;
        }
    return true;
}

private static boolean ok(char[][] b, int r, int c, char ch) {
    for (int k = 0; k < 9; k++) {
        if (b[r][k] == ch || b[k][c] == ch) return false;
        if (b[3 * (r / 3) + k / 3][3 * (c / 3) + k % 3] == ch) return false;
    }
    return true;
}

    List<List<Integer>> adj = new ArrayList<>();
    for (int i = 0; i < n; i++) adj.add(new ArrayList<>());
    for (int[] e : edges) { adj.get(e[0]).add(e[1]); adj.get(e[1]).add(e[0]); }

}

private static boolean color(int v, int n, int m, List<List<Integer>> adj, int[] col) {
    if (v == n) return true;
    for (int c = 1; c <= m; c++) {
        boolean ok = true;
        for (int u : adj.get(v)) if (col[u] == c) { ok = false; break; }
        if (!ok) continue;
        col[v] = c;
        if (color(v + 1, n, m, adj, col)) return true;
        col[v] = 0;
    }
    return false;
}
