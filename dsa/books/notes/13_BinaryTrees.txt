private static int[] go(TreeNode n, int[] best) {
    if (n == null) return new int[]{Integer.MAX_VALUE, Integer.MIN_VALUE, 0};
    int[] l = go(n.left, best), r = go(n.right, best);
    if (l[1] < n.val && n.val < r[0]) {
        int size = l[2] + r[2] + 1;
        best[0] = Math.max(best[0], size);
        return new int[]{Math.min(l[0], n.val), Math.max(r[1], n.val), size};
    }
    return new int[]{Integer.MIN_VALUE, Integer.MAX_VALUE, Math.max(l[2], r[2])};
