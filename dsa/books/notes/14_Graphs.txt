    int n = graph.length;
    int[] col = new int[n];
    Arrays.fill(col, -1);
    for (int s = 0; s < n; s++) {
        if (col[s] != -1) continue;
        Queue<Integer> q = new LinkedList<>();
        q.add(s);
        col[s] = 0;
        while (!q.isEmpty()) {
            int u = q.poll();
            for (int v : graph[u]) {
                if (col[v] == -1) { col[v] = 1 - col[u]; q.add(v); }
                else if (col[v] == col[u]) return false;
            }
        }
    }
    return true;
}

            for (int i = 0; i <= n; i++) { parent[i] = i; size[i] = 1; }

            if (pa == pb) return;

            if (size[pa] < size[pb]) { parent[pa] = pb; size[pb] += size[pa]; }
            else { parent[pb] = pa; size[pa] += size[pb]; }
        }
    }

    d.unionByRank(2, 3);

    }

}

public static List<Integer> articulationPoints(int n, List<List<Integer>> adj) {
    boolean[] vis = new boolean[n], mark = new boolean[n];
    int[] tin = new int[n], low = new int[n];
    for (int i = 0; i < n; i++) if (!vis[i]) dfs(i, -1, adj, tin, low, vis, mark, new int[]{0});

private static void dfs(int u, int p, List<List<Integer>> adj, int[] tin, int[] low, boolean[] vis, boolean[] mark, int[] timer) {
    vis[u] = true;
    tin[u] = low[u] = timer[0]++;
    int children = 0;
    for (int v : adj.get(u)) {
        if (v == p) continue;
        if (vis[v]) low[u] = Math.min(low[u], tin[v]);
        else {
            dfs(v, u, adj, tin, low, vis, mark, timer);
            low[u] = Math.min(low[u], low[v]);
            if (low[v] >= tin[u] && p != -1) mark[u] = true;
            children++;
        }
    }
    if (p == -1 && children > 1) mark[u] = true;
}
