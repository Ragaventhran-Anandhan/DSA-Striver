why: mid maps to row mid/m and column mid%m since rows chain in order.
expl: lo = 0, hi = n*m-1 | mid element = matrix[mid/m][mid%m] | Narrow based on comparison

public static boolean searchMatrix(int[][] matrix, int target) {
    int n = matrix.length, m = matrix[0].length, lo = 0, hi = n * m - 1;
    while (lo <= hi) {
        int mid = (lo + hi) >>> 1, v = matrix[mid / m][mid % m];
        if (v == target) return true;
        if (v < target) lo = mid + 1; else hi = mid - 1;
    }
    return false;
}

public static List<Integer> majorityElementII(int[] arr) {
    int c1 = 0, c2 = 1, n1 = 0, n2 = 0;
    for (int x : arr) {
        if (x == c1) n1++;
        else if (x == c2) n2++;
        else if (n1 == 0) { c1 = x; n1 = 1; }
        else if (n2 == 0) { c2 = x; n2 = 1; }
        else { n1--; n2--; }
    }
    n1 = n2 = 0;
    for (int x : arr) { if (x == c1) n1++; else if (x == c2) n2++; }
    List<Integer> r = new ArrayList<>();
    if (n1 > arr.length / 3) r.add(c1);
    if (n2 > arr.length / 3) r.add(c2);
    return r;
}

why: Subarray XOR equals prefix[j] ^ prefix[i]; need prefix[i] = prefix[j] ^ k.

public static int subarraysWithXorK(int[] arr, int k) {
    Map<Integer, Integer> m = new HashMap<>();
    m.put(0, 1);
    int xr = 0, count = 0;
    for (int x : arr) {
        xr ^= x;
        count += m.getOrDefault(xr ^ k, 0);
        m.merge(xr, 1, Integer::sum);
    }
    return count;
}

public static int[] repeatingAndMissing(int[] arr) {
    int n = arr.length;
    int[] cnt = new int[n + 1];
    for (int x : arr) cnt[x]++;
    int rep = -1, miss = -1;
    for (int i = 1; i <= n; i++) {
        if (cnt[i] == 2) rep = i;
        else if (cnt[i] == 0) miss = i;
    }
    return new int[]{rep, miss};
}

public static long countInversions(int[] arr) {
    return sort(arr.clone(), 0, arr.length - 1);
}

private static long sort(int[] a, int lo, int hi) {
    if (lo >= hi) return 0;
    int mid = (lo + hi) / 2;
    long c = sort(a, lo, mid) + sort(a, mid + 1, hi);
    int[] t = new int[hi - lo + 1];
    int i = lo, j = mid + 1, k = 0;
    while (i <= mid && j <= hi) {
        if (a[i] <= a[j]) t[k++] = a[i++];
        else { c += mid - i + 1; t[k++] = a[j++]; }
    }
    while (i <= mid) t[k++] = a[i++];
    while (j <= hi) t[k++] = a[j++];
    System.arraycopy(t, 0, a, lo, t.length);
    return c;
}

private static int sort(int[] a, int lo, int hi) {
    if (lo >= hi) return 0;
    int mid = (lo + hi) / 2;
    int c = sort(a, lo, mid) + sort(a, mid + 1, hi);
    int j = mid + 1;
    for (int i = lo; i <= mid; i++) {
        while (j <= hi && (long) a[i] > 2L * a[j]) j++;
        c += j - (mid + 1);
    }
    Arrays.sort(a, lo, hi + 1);
    return c;
}

brute: Multiply every subarray, O(n^2).
opt: Track both max and min product ending here.
why: A negative number can turn the minimum into the maximum.
expl: curMax, curMin, best = arr[0] | On negative swap max and min | Update with x or x*prev | best = max(best, curMax)

public static int maxProductSubarray(int[] arr) {
    int max = arr[0], min = arr[0], best = arr[0];
    for (int i = 1; i < arr.length; i++) {
        int x = arr[i];
        if (x < 0) { int t = max; max = min; min = t; }
        max = Math.max(x, max * x);
        min = Math.min(x, min * x);
        best = Math.max(best, max);
    }
    return best;
}
